function createMonthlySheet(monthName, daysInMonth) {
  const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  
  // Create or get the sheet for this month
  let sheet = spreadsheet.getSheetByName(monthName);
  if (sheet) {
    sheet.clear();
    // Remove all existing protections
    const protections = sheet.getProtections(SpreadsheetApp.ProtectionType.RANGE);
    for (let i = 0; i < protections.length; i++) {
      protections[i].remove();
    }
  } else {
    sheet = spreadsheet.insertSheet(monthName);
  }
  
  // Set column widths
  sheet.setColumnWidth(1, 50);  // Empty column A
  sheet.setColumnWidth(2, 50);  // Empty column B
  sheet.setColumnWidth(3, 200); // Date
  sheet.setColumnWidth(4, 130); // Amount
  sheet.setColumnWidth(5, 280); // Note
  sheet.setColumnWidth(6, 150); // Day Limit
  sheet.setColumnWidth(7, 150); // Days to Survive
  
  // MAIN HEADER
  sheet.getRange('C1:G1').merge().setValue('💰 MONTHLY BUDGET TRACKER')
    .setFontSize(24).setFontWeight('bold').setHorizontalAlignment('center')
    .setBackground('#1A1A2E').setFontColor('#FFFFFF')
    .setVerticalAlignment('middle');
  sheet.setRowHeight(1, 50);
  
  // Month Input Section
  sheet.getRange('C3').setValue('📅 Month:').setFontWeight('bold').setFontSize(12);
  sheet.getRange('D3:E3').merge().setValue(monthName)
    .setFontSize(14).setFontWeight('bold')
    .setBackground('#E8F5E9').setHorizontalAlignment('center')
    .setBorder(true, true, true, true, true, true, '#4CAF50', SpreadsheetApp.BorderStyle.SOLID_MEDIUM);
  
  // Monthly Budget Input
  sheet.getRange('C5').setValue('💵 Monthly Budget:').setFontWeight('bold').setFontSize(12);
  sheet.getRange('D5').setValue(0)
    .setBackground('#2196F3').setFontColor('#FFFFFF')
    .setFontWeight('bold').setFontSize(16)
    .setNumberFormat('"Rs "#,##0.00')
    .setHorizontalAlignment('center')
    .setBorder(true, true, true, true, true, true, '#1976D2', SpreadsheetApp.BorderStyle.SOLID_THICK);
  sheet.setRowHeight(5, 40);
  
  // Fixed Expenses Header
  sheet.getRange('C7:D7').merge().setValue('📋 FIXED MONTHLY EXPENSES')
    .setFontWeight('bold').setFontSize(11)
    .setBackground('#607D8B').setFontColor('#FFFFFF')
    .setHorizontalAlignment('center');
  
  const expenses = ['🏠 Bodim Fees', '🛡️ CoverUps', '💰 Savings', '📌 Option 1', '📌 Option 2'];
  const defaultValues = [0, 0, 0, 0, 0];
  
  for (let i = 0; i < expenses.length; i++) {
    sheet.getRange(8 + i, 3).setValue(expenses[i]).setFontWeight('bold');
    sheet.getRange(8 + i, 4).setValue(defaultValues[i])
      .setNumberFormat('"Rs "#,##0.00')
      .setBackground('#F5F5F5')
      .setHorizontalAlignment('right');
  }
  
  // Summary Calculations
  sheet.getRange('C14').setValue('📊 Total Fixed:').setFontWeight('bold').setFontSize(11);
  sheet.getRange('D14').setFormula('=SUM(D8:D12)')
    .setBackground('#EF5350').setFontColor('#FFFFFF')
    .setFontWeight('bold').setNumberFormat('"Rs "#,##0.00')
    .setHorizontalAlignment('center')
    .setBorder(true, true, true, true, true, true, '#C62828', SpreadsheetApp.BorderStyle.SOLID_MEDIUM);
  
  sheet.getRange('C15').setValue('✅ Variable Budget:').setFontWeight('bold').setFontSize(12);
  sheet.getRange('D15').setFormula('=D5-D14')
    .setBackground('#66BB6A').setFontColor('#FFFFFF')
    .setFontWeight('bold').setFontSize(16).setNumberFormat('"Rs "#,##0.00')
    .setHorizontalAlignment('center')
    .setBorder(true, true, true, true, true, true, '#388E3C', SpreadsheetApp.BorderStyle.SOLID_THICK);
  sheet.setRowHeight(15, 40);
  
  sheet.getRange('C16').setValue('📅 Per Day Limit (÷' + daysInMonth + '):').setFontWeight('bold');
  sheet.getRange('D16').setFormula('=D15/' + daysInMonth)
    .setBackground('#FFA726').setFontColor('#FFFFFF')
    .setFontWeight('bold').setNumberFormat('"Rs "#,##0.00')
    .setHorizontalAlignment('center')
    .setBorder(true, true, true, true, true, true, '#F57C00', SpreadsheetApp.BorderStyle.SOLID_MEDIUM);
  
  // Current Balance Section (Right side)
  sheet.getRange('F5:G5').merge().setValue('💳 CURRENT BALANCE')
    .setFontWeight('bold').setFontSize(12)
    .setBackground('#00897B').setFontColor('#FFFFFF')
    .setHorizontalAlignment('center')
    .setVerticalAlignment('middle');
  
  sheet.getRange('F6:G6').merge().setFormula('=D15+SUM(D20:D)')
    .setBackground('#26A69A').setFontColor('#FFFFFF')
    .setFontWeight('bold').setFontSize(20)
    .setHorizontalAlignment('center').setNumberFormat('"Rs "#,##0.00')
    .setVerticalAlignment('middle')
    .setBorder(true, true, true, true, true, true, '#00796B', SpreadsheetApp.BorderStyle.SOLID_THICK);
  sheet.setRowHeight(6, 50);
  
  // Survival Days Display
  sheet.getRange('F14:G14').merge().setValue('⏳ Days Can Survive')
    .setFontWeight('bold').setFontSize(11)
    .setBackground('#7E57C2').setFontColor('#FFFFFF')
    .setHorizontalAlignment('center');
  
  sheet.getRange('F15:G15').merge()
    .setFormula('=IF($D$16=0,"N/A",($D$15+SUM(D20:D))/$D$16)')
    .setBackground('#9575CD').setFontColor('#FFFFFF')
    .setFontWeight('bold').setFontSize(18)
    .setNumberFormat('0.0 " days"')
    .setHorizontalAlignment('center')
    .setVerticalAlignment('middle')
    .setBorder(true, true, true, true, true, true, '#5E35B1', SpreadsheetApp.BorderStyle.SOLID_THICK);
  
  // Transaction Table Headers
  const headers = ['DATE', 'AMOUNT', 'NOTE', 'DAY LIMIT', 'DAYS TO SURVIVE'];
  sheet.getRange('C18:G18').setValues([headers])
    .setFontWeight('bold').setBackground('#37474F').setFontColor('#FFFFFF')
    .setHorizontalAlignment('center').setFontSize(11)
    .setVerticalAlignment('middle')
    .setBorder(true, true, true, true, true, true, '#263238', SpreadsheetApp.BorderStyle.SOLID_THICK);
  sheet.setRowHeight(18, 35);
  
  // Add helper text row
  sheet.getRange('C19').setValue('Select Date').setFontStyle('italic').setFontSize(9).setFontColor('#999999');
  sheet.getRange('D19').setValue('Enter amount').setFontStyle('italic').setFontSize(9).setFontColor('#999999');
  sheet.getRange('E19').setValue('Add note (optional)').setFontStyle('italic').setFontSize(9).setFontColor('#999999');
  sheet.getRange('F19').setValue('Auto-calculated').setFontStyle('italic').setFontSize(9).setFontColor('#999999');
  sheet.getRange('G19').setValue('Auto-calculated').setFontStyle('italic').setFontSize(9).setFontColor('#999999');
  sheet.getRange('C19:G19').setBackground('#FAFAFA');
  
  // First data row formulas (row 20)
  sheet.getRange('F20').setFormula('=IF(C20="","",IF(C20=C19,F19+D20,$D$16+D20))');
  sheet.getRange('G20').setFormula('=IF(C20="","",' + daysInMonth + '-DAY(C20))');
  
  // Copy formulas down
  sheet.getRange('F20:G20').copyTo(sheet.getRange('F21:G1019'));
  
  // Format numbers
  sheet.getRange('D20:D').setNumberFormat('"Rs "#,##0.00');
  sheet.getRange('F20:F').setNumberFormat('"Rs "#,##0.00');
  sheet.getRange('G20:G').setNumberFormat('0.0 " days"');
  
  // Add borders
  sheet.getRange('C20:G1019').setBorder(true, true, true, true, true, true, '#E0E0E0', SpreadsheetApp.BorderStyle.SOLID);
  
  // Conditional formatting
  const dayLimitRange = sheet.getRange('F20:F');
  const rule1 = SpreadsheetApp.newConditionalFormatRule()
    .whenNumberLessThan(0)
    .setBackground('#FFCDD2')
    .setFontColor('#C62828')
    .setRanges([dayLimitRange])
    .build();
  const rule2 = SpreadsheetApp.newConditionalFormatRule()
    .whenNumberGreaterThanOrEqualTo(0)
    .setBackground('#E1F5FE')
    .setRanges([dayLimitRange])
    .build();
  
  const survivalRange = sheet.getRange('G20:G');
  const rule3 = SpreadsheetApp.newConditionalFormatRule()
    .whenNumberLessThan(7)
    .setBackground('#FFCDD2')
    .setFontColor('#C62828')
    .setBold(true)
    .setRanges([survivalRange])
    .build();
  const rule4 = SpreadsheetApp.newConditionalFormatRule()
    .whenNumberBetween(7, 13.99)
    .setBackground('#FFF9C4')
    .setFontColor('#F57F17')
    .setBold(true)
    .setRanges([survivalRange])
    .build();
  const rule5 = SpreadsheetApp.newConditionalFormatRule()
    .whenNumberGreaterThanOrEqualTo(14)
    .setBackground('#C8E6C9')
    .setFontColor('#2E7D32')
    .setBold(true)
    .setRanges([survivalRange])
    .build();
  
  const rules = sheet.getConditionalFormatRules();
  rules.push(rule1, rule2, rule3, rule4, rule5);
  sheet.setConditionalFormatRules(rules);
  
  // Freeze rows and columns
  sheet.setFrozenRows(19);
  sheet.setFrozenColumns(2);
  
  // Add data validation
  const dateRange = sheet.getRange('C20:C');
  const dateValidation = SpreadsheetApp.newDataValidation()
    .requireDate()
    .setAllowInvalid(true)
    .setHelpText('Enter date in format: MM/DD/YYYY')
    .build();
  dateRange.setDataValidation(dateValidation);
  
  // Align columns
  sheet.getRange('C20:C').setHorizontalAlignment('center');
  sheet.getRange('D20:D').setHorizontalAlignment('right');
  sheet.getRange('F20:G').setHorizontalAlignment('center');
  
  // ========================================
  // 🔒 PROTECT RANGES WITH WARNINGS
  // ========================================
  
  // Protection 1: Header and Month Section (F5:G6)
  const protection1 = sheet.getRange('C1:G7').protect();
  protection1.setDescription('⚠️ Header & Month Name - Contains important labels');
  protection1.setWarningOnly(true);
  
  // Protection 2: Summary Section (C14:G19)
  const protection2 = sheet.getRange('C14:G19').protect();
  protection2.setDescription('⚠️ Summary & Headers - Contains formulas and labels');
  protection2.setWarningOnly(true);
  
  // Protection 3: Expense Labels (C8:C12)
  // const protection3 = sheet.getRange('C8:C12').protect();
  // protection3.setDescription('⚠️ Expense Labels - Fixed expense names');
  // protection3.setWarningOnly(true);
  
  // Protection 4: Auto-calculated Columns (F20:G1000)
  const protection4 = sheet.getRange('F20:G1000').protect();
  protection4.setDescription('⚠️ Auto-Calculated - Day Limit & Days to Survive formulas');
  protection4.setWarningOnly(true);
}

function getMonthNumber(monthName) {
  const months = {
    'January': 0, 'February': 1, 'March': 2, 'April': 3,
    'May': 4, 'June': 5, 'July': 6, 'August': 7,
    'September': 8, 'October': 9, 'November': 10, 'December': 11
  };
  
  for (const month in months) {
    if (monthName.includes(month)) {
      return months[month];
    }
  }
  return 0;
}

function clearSampleData() {
  const sheet = SpreadsheetApp.getActiveSheet();
  
  const ui = SpreadsheetApp.getUi();
  const response = ui.alert(
    '🗑️ Clear Sample Data',
    'This will reset all values to 0 and remove all transactions from ' + sheet.getName() + '.\n\nAre you sure?',
    ui.ButtonSet.YES_NO
  );
  
  if (response == ui.Button.YES) {
    sheet.getRange('D5').setValue(0);
    for (let i = 0; i < 5; i++) {
      sheet.getRange(8 + i, 4).setValue(0);
    }
    sheet.getRange('C20:E1019').clearContent();
    
    ui.alert('✅ Sample Data Cleared from ' + sheet.getName() + '!');
  }
}
